class Solution {
    public int[] solution(int[] arr, int n) {
        int[] answer = new int[arr.length];
        if (answer.length % 2 == 1) {
            for(int i = 0; i < arr.length; i++) {
                if (i % 2 == 0) {
                    answer[i] = arr[i] + n;
                } else {
                  answer[i] = arr[i];
                }
            }
        } else {
            for (int i = 0; i < arr.length; i++) {
                if (i % 2 == 1) {
                    answer[i] = arr[i] + n;
                } else {
                  answer[i] = arr[i];
                }
            }
        }
        return answer;
    }
}

public class Main {
    public static void main(String[] args) {
        int[] arr = {1, 2, 3};
        int n = 3;
        Solution u = new Solution();
        int[] arr2 = u.solution(arr, n);
        System.out.println(Arrays.toString(arr2));
    }
}
