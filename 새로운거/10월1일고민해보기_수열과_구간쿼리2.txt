import java.util.*;

class Solution {
    public int[] solution(int[] arr, int[][] queries) {
        int[] answer = {};
        for(int i = 0; i < queries.length; i++) {
            int neww = 0;
            int newww = 1000000;
            for (int k = 0; k < arr.length; k++) {
                
                if (arr[k] > queries[i][2] && arr[k] <= queries[i][1] && arr[k] >= queries[i][0] && arr[k] < newww) {
                    newww = arr[k];
                }

            }
            neww = 0;
        }
        return answer;
    }
}

public class Main {
    public static void main(String[] args) {
      System.out.println("Hello, World!");
    }
}
