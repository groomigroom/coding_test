                              .collect(Collectors.toCollection(ArrayList::new));
        for(int i = 0; i < queries.length; i++) {
            int neww = 0;
            int newww = 1000000;
            for (int k = 0; k < arr.length; k++) {
                
                if (arr[k] > queries[i][2] && arr[k] <= queries[i][1] && arr[k] >= queries[i][0] && arr[k] < newww) {
                    newww = arr[k];
                }
                neww = newww;
            }
            answer.add(neww);
            neww = 0;
        }
        return answer;
    }
