def solution(n, control):
    answer = 0
    for i in range (0, len(control)) {
        switch (control.charAt(i)) {
            case 'w':
                n++;
                break;
            case 's':
                n--;
                break;
            case 'd':
                n += 10;
                break;
            case 'a':
                n -= 10;
                break;
        }
    }
        answer = n;
    return answer
