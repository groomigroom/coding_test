// 1. 0부터 14까지 숫자가 순서대로 들어있는 배열 생성 ([0, 1, 2, ..., 14])
const number_random_arr = Array.from({ length: 15 }, (_, i) => i);

// 2. 배열을 무작위로 뒤섞기 (피셔-예이츠 셔플 알고리즘)
for (let i = number_random_arr.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [number_random_arr[i], number_random_arr[j]] = [number_random_arr[j], number_random_arr[i]];
}

console.log(number_random_arr); // 중복 없는 무작위 15개 숫자 배열 출력
