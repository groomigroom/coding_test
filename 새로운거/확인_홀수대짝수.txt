def solution(num_list):
    answer = 0
    hap1 = 0
    hap2 = 0
    for i in range(0, len(num_list), 2):
        hap1 += num_list[i]
    for j in range(1, len(num_list), 2):
        hap2 += num_list[i]
    if hap1 >= hap2:
        answer = hap1
    else:
        answer = hap2
    return answer
