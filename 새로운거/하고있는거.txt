class Solution {
    public int solution(String my_string, String is_prefix) {
        int answer = 0;
        for(int i = 0; i < is_prefix.length(); i++) {
            int count = 0;
            if (is_prefix.charAt(i) == my_string.charAt(i)) {
                count++;
            }
            if (count == is_prefix.length()) {
                answer = 1;
            } else {
                answer = 0;
            }
        }
        return answer;
    }
}
