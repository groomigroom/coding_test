import java.util.*;

class Solution {
    public String[] solution(String my_string) {
        List<String> str_list = new ArrayList<>();
        int j = 0;
        String stt;
        for (int i = 0; i < my_string.length(); i++) {
            if(my_string.charAt(i) == ' ') {
                stt = my_string.substring(j, i);
                j = i + 1;
                str_list.add(stt);
            }
        }

        if (j < my_string.length()) {
            str_list.add(my_string.substring(j));
        }

        String[] answer = str_list.toArray(new String[0]);
        return answer;
    }
}

public class Main {
    public static void main(String[] args) {
      String my_string = "kim groom ii";
      Solution sol = new Solution();
      String[] solsol = sol.solution(my_string);
      System.out.println(Arrays.toString(solsol));
    }
}
