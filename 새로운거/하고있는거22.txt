    import java.util.ArrayList; // 💡 ArrayList 사용을 위해 임포트 필요 [3]
import java.util.List;

class Solution {
    public int[] solution(int[] arr) {
        // 크기가 가변적인 리스트를 스택(stk) 대용으로 생성합니다.
        List<Integer> stkList = new ArrayList<>();
        
        int i = 0;
        while (i < arr.length) {
            // 1. stk가 비어있으면 arr[i]를 추가하고 i를 1 증가시킵니다.
            if (stkList.isEmpty()) {
                stkList.add(arr[i]); // 👈 파이썬의 append와 동일 [3]
                i++;
            } 
            // 2. stk에 원소가 있고, 마지막 원소가 arr[i]보다 작으면 arr[i]를 추가하고 i를 1 증가시킵니다.
            else if (stkList.get(stkList.size() - 1) < arr[i]) {
                stkList.add(arr[i]); // 👈 append [3]
                i++;
            } 
            // 3. stk에 원소가 있고, 마지막 원소가 arr[i]보다 크거나 같으면 마지막 원소를 제거합니다.
            else {
                stkList.remove(stkList.size() - 1); // 👈 파이썬의 pop()과 동일
            }
        }
        
        // 결과 반환을 위해 List<Integer>를 int[] 배열로 변환합니다.
        int[] stk = new int[stkList.size()];
        for (int j = 0; j < stkList.size(); j++) {
            stk[j] = stkList.get(j);
        }
        
        return stk;
    }
}
