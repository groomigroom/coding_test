class Solution {
    public int solution(int a, int b) {
        int answer = 0;
        String str = "";
        int intab = Integer.parseInt(str + a + b);
        int intba = Integer.parseInt(str + b + a);
        if (intab > intba) {
            answer = intab;
        } else {
            answer = intba;
        }
        return answer;
    }
}
