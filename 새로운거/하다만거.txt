import java.util.*;

class Solution {
    public String solution(String rny_string) {
        String answer = "";
        for (int i = 0; i < rny_string.length; i++) {
        if (rny_string.charAt(i) == "m") {
            answer += "rn";
        } else {
            answer += rny_string[i];
        }
    }
        return answer;
    }
}

public class Main {
    public static void main(String[] args) {
      System.out.println("Hello, World!");
    }
}
